//Q5.  GCD using Euclid's algorithm

#include <stdio.h>

int main()
{
    int a, b, temp;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    while (b != 0)
    {
        temp = b;
        b = a % b;
        a = temp;
    }

    printf("GCD = %d", a);

    return 0;
}

// Q2. Count relatively prime values

#include <stdio.h>

int gcd(int a, int b)
{
    while (b != 0)
    {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int main()
{
    int n, x, count = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int a[n];

    printf("Enter array elements: ");
    for (int i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter a number: ");
    scanf("%d", &x);

    for (int i = 0; i < n; i++)
    {
        if (gcd(a[i], x) == 1)
        {
            count++;
        }
    }

    printf("Count = %d", count);

    return 0;
}

//Q4. Pairs with Equal Digit Sums

#include <stdio.h>

int digitSum(int n)
{
    int sum = 0;

    if (n < 0)
        n = -n;

    while (n > 0)
    {
        sum += n % 10;
        n = n / 10;
    }

    return sum;
}

int main()
{
    int n, i, j;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int a[n];

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (i = 0; i < n; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (digitSum(a[i]) == digitSum(a[j]))
            {
                printf("(%d, %d)\n", a[i], a[j]);
            }
        }
    }

    return 0;
}
// Q5. Closest Pair Difference

#include <stdio.h>
#include <stdlib.h>

int main()
{
    int n, i, j, min, x, y;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int a[n];

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    min = abs(a[0] - a[1]);
    x = a[0];
    y = a[1];

    for (i = 0; i < n; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (abs(a[i] - a[j]) < min)
            {
                min = abs(a[i] - a[j]);
                x = a[i];
                y = a[j];
            }
        }
    }

    printf("Closest pair: %d %d\n", x, y);
    printf("Minimum difference = %d", min);

    return 0;
}

// Q1. Count Vowels, Consonants, Digits and Spaces

#include <stdio.h>

int main()
{
    char str[100];
    int i, v = 0, c = 0, d = 0, s = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] >= '0' && str[i] <= '9')
            d++;
        else if (str[i] == ' ')
            s++;
        else if ((str[i] >= 'a' && str[i] <= 'z') ||
                 (str[i] >= 'A' && str[i] <= 'Z'))
        {
            if (str[i] == 'a' || str[i] == 'e' ||
                str[i] == 'i' || str[i] == 'o' ||
                str[i] == 'u' || str[i] == 'A' ||
                str[i] == 'E' || str[i] == 'I' ||
                str[i] == 'O' || str[i] == 'U')
                v++;
            else
                c++;
        }
    }

    printf("Vowels = %d\n", v);
    printf("Consonants = %d\n", c);
    printf("Digits = %d\n", d);
    printf("Spaces = %d\n", s);

    return 0;
}

//Q2. Run-Length Encode

#include <stdio.h>

int main()
{
    char str[100];
    int i, count;

    printf("Enter a string: ");
    scanf("%99s", str);

    for (i = 0; str[i] != '\0'; i++)
    {
        count = 1;

        while (str[i] == str[i + 1])
        {
            count++;
            i++;
        }

        printf("%c%d", str[i], count);
    }

}

// Q3 — Find a Saddle Point

#include <stdio.h>

int main()
{
    int a[10][10], r, c, i, j;
    int k, min, flag = 0;

    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);

    printf("Enter matrix elements:\n");

    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
            scanf("%d", &a[i][j]);
    }

    for (i = 0; i < r; i++)
    {
        min = a[i][0];
        k = 0;

        for (j = 1; j < c; j++)
        {
            if (a[i][j] < min)
            {
                min = a[i][j];
                k = j;
            }
        }

        for (j = 0; j < r; j++)
        {
            if (a[j][k] > min)
                break;
        }

        if (j == r)
        {
            printf("Saddle point = %d", min);
            flag = 1;
        }
    }

    if (flag == 0)
        printf("No saddle point found");

    return 0;
}

//Q4. Count Local Peaks

#include <stdio.h>

int main()
{
    int a[10][10], r, c;
    int i, j, x, y, count = 0;
    int flag;

    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);

    printf("Enter matrix elements:\n");

    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
            scanf("%d", &a[i][j]);
    }

    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            flag = 1;

            for (x = i - 1; x <= i + 1; x++)
            {
                for (y = j - 1; y <= j + 1; y++)
                {
                    if (x >= 0 && x < r &&
                        y >= 0 && y < c &&
                        (x != i || y != j))
                    {
                        if (a[i][j] <= a[x][y])
                            flag = 0;
                    }
                }
            }

            if (flag == 1)
                count++;
        }
    }

    printf("Number of local peaks = %d", count);

    return 0;
}
